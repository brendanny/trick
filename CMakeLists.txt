cmake_minimum_required(VERSION 3.26)

# Resolve aliases too: an in-source configure can overwrite the legacy Makefile.
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" trick_source_dir)
file(REAL_PATH "${CMAKE_CURRENT_BINARY_DIR}" trick_binary_dir)
if(trick_source_dir STREQUAL trick_binary_dir)
    message(FATAL_ERROR
        "Trick requires an out-of-source build. Use cmake -S <source> -B <build>. "
        "Remove only CMakeCache.txt and CMakeFiles created by this failed attempt; "
        "do not remove Trick's checked-in Makefile.")
endif()

include(cmake/TrickVersion.cmake)
trick_read_version("${CMAKE_CURRENT_SOURCE_DIR}/share/trick/trick_ver.txt")
project(Trick VERSION "${TRICK_VERSION_NUMBER}" LANGUAGES C CXX)

option(TRICK_BUILD_ICG "Build the existing native interface code generator" OFF)

option(TRICK_BUILD_MEMORY_MANAGER "Build the native memory manager and parsers" OFF)

option(TRICK_BUILD_INTEGRATORS "Build integration libraries" OFF)
option(TRICK_USE_ER7_UTILS "Use bundled ER7 integration algorithms" ON)
option(TRICK_USE_ER7_CHECKPOINTHELPER "Use an external ER7 CheckpointHelper variant" OFF)
if(TRICK_USE_ER7_CHECKPOINTHELPER)
    message(FATAL_ERROR "CheckpointHelper is not bundled. External ER7 variants need a reviewed target contract; this preview does not silently select one.")
endif()
option(TRICK_BUILD_CORE_METADATA "Generate and compile core ICG metadata" OFF)
option(TRICK_BUILD_CORE "Build native core runtime services" OFF)
option(TRICK_BUILD_PYTHON "Build the embedded Python input processor and SWIG modules" OFF)
option(TRICK_BUILD_SDK "Build a simulation SDK using the existing Trick simulation tools" OFF)
if(TRICK_BUILD_SDK)
    set(TRICK_BUILD_PYTHON ON)
endif()
if(TRICK_BUILD_PYTHON)
    set(TRICK_BUILD_CORE ON)
endif()
if(TRICK_BUILD_CORE)
    set(TRICK_BUILD_CORE_METADATA ON)
    set(TRICK_BUILD_UTILITIES ON)
endif()
if(TRICK_BUILD_CORE_METADATA)
    set(TRICK_BUILD_INTEGRATORS ON)
    set(TRICK_BUILD_ICG ON)
endif()
if(TRICK_BUILD_INTEGRATORS)
    set(TRICK_BUILD_MEMORY_MANAGER ON)
endif()

include(cmake/TrickCompilers.cmake)
include(cmake/TrickDependencies.cmake)

configure_file(cmake/TrickVersionInfo.cmake.in TrickVersionInfo.cmake @ONLY)

include(CTest)
option(TRICK_BUILD_UTILITIES "Build the native utility archives" ON)
option(TRICK_BUILD_UTILITY_TESTS "Build focused utility tests (requires GoogleTest)" OFF)
if(TRICK_BUILD_UTILITY_TESTS AND (NOT BUILD_TESTING OR NOT TRICK_BUILD_UTILITIES))
    message(FATAL_ERROR "Utility tests require BUILD_TESTING and TRICK_BUILD_UTILITIES")
endif()
if(TRICK_BUILD_UTILITIES)
    add_subdirectory(trick_source/trick_utils)
endif()
if(TRICK_BUILD_ICG)
    add_subdirectory(trick_source/codegen/Interface_Code_Gen)
endif()
if(TRICK_BUILD_MEMORY_MANAGER)
    add_subdirectory(trick_source/sim_services)
endif()
if(TRICK_BUILD_SDK)
    include(cmake/TrickSDK.cmake)
endif()
if(BUILD_TESTING)
    add_subdirectory(cmake/tests)
endif()

message(STATUS "Trick ${TRICK_VERSION}: CMake core SDK preview (C3)")
if(TRICK_BUILD_SDK)
    include(cmake/TrickInstall.cmake)
    include(cmake/TrickPackage.cmake)
else()
    # Do not turn a successful utility build into a misleading empty SDK install.
    install(CODE [[
        message(FATAL_ERROR "Trick SDK installation requires -DTRICK_BUILD_SDK=ON.")
    ]])
endif()
